cmake_minimum_required(VERSION 3.20)
project(CompatLoL LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
option(COMPATLOL_SANITIZERS "Enable host sanitizers" OFF)
add_library(compatlol_core src/macho.cpp src/host.cpp)
target_include_directories(compatlol_core PUBLIC src third_party/json)
target_compile_options(compatlol_core PRIVATE -Wall -Wextra -Wpedantic)
if(COMPATLOL_SANITIZERS)
  target_compile_options(compatlol_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(compatlol_core PUBLIC -fsanitize=address,undefined)
endif()
add_executable(compatlol tools/main.cpp)
target_link_libraries(compatlol PRIVATE compatlol_core)
if(NOT APPLE)
  find_package(OpenSSL REQUIRED)
  target_link_libraries(compatlol PRIVATE OpenSSL::Crypto)
endif()
if(BUILD_TESTING)
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND)
    add_test(NAME audit-harness COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/audit_harness_tests.py)
    set_tests_properties(audit-harness PROPERTIES TIMEOUT 30)
  else()
    message(WARNING "Python3 unavailable: audit-harness test not registered")
  endif()
  add_executable(host_tests tests/host_tests.cpp)
  target_link_libraries(host_tests PRIVATE compatlol_core)
  add_test(NAME host-contracts COMMAND host_tests)
  add_test(NAME doctor COMMAND compatlol doctor --json)
  add_executable(diagnostics_tests tests/diagnostics_tests.cpp)
  target_link_libraries(diagnostics_tests PRIVATE compatlol_core)
  add_test(NAME diagnostics-contracts COMMAND diagnostics_tests)
  add_test(NAME cli-contracts COMMAND ${CMAKE_COMMAND}
    -DCLI=$<TARGET_FILE:compatlol> -P ${CMAKE_SOURCE_DIR}/tests/cli.cmake)
  add_test(NAME riot-denied COMMAND compatlol run --profile riot --bundle nonexistent)
  set_tests_properties(riot-denied PROPERTIES WILL_FAIL TRUE)
endif()
option(COMPATLOL_BUILD_ABI_FIXTURE "Build open native reference fixture" ON)
if(COMPATLOL_BUILD_ABI_FIXTURE)
  find_package(Threads REQUIRED)
  add_executable(abi_fixture pocs/abi/main.cpp)
  target_link_libraries(abi_fixture PRIVATE Threads::Threads)
  if(BUILD_TESTING)
    add_test(NAME native-reference-abi COMMAND abi_fixture)
    add_test(NAME inspect-read-only COMMAND ${CMAKE_COMMAND}
      -DCLI=$<TARGET_FILE:compatlol> -DFIXTURE=$<TARGET_FILE:abi_fixture>
      -DTEST_DIR=${CMAKE_BINARY_DIR}/inspect-fixture -P ${CMAKE_SOURCE_DIR}/tests/inspect.cmake)
  endif()
endif()
install(TARGETS compatlol RUNTIME DESTINATION bin)
